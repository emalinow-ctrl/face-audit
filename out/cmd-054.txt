            })

            rel = parse_qs(parsed.query).get("img", [""])[0]

            ctype = mimetypes.guess_type(str(p))[0] or "image/jpeg"

            p = lf["path"].resolve()
            if self.ctx.out_dir.resolve() not in p.parents:
                return self._send(403, b"forbidden", "text/plain")
            ctype = mimetypes.guess_type(str(p))[0] or "image/jpeg"
